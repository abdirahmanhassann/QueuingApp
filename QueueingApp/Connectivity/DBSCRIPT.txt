/* Copy legacy dbo.Shop → dbo.Shops, preserving ids that Users.ShopId points at. */

BEGIN
    DELETE s
    FROM dbo.Shops s
    WHERE s.QrCode IN (N'fade-room', N'atelier-noir')
      AND NOT EXISTS (SELECT 1 FROM dbo.Shop x WHERE x.QrCode = s.QrCode OR x.Name = s.Name)
      AND (
            OBJECT_ID(N'dbo.Users', N'U') IS NULL
            OR NOT EXISTS (SELECT 1 FROM dbo.Users u WHERE u.ShopId = s.Id)
          )
      AND (
            OBJECT_ID(N'dbo.QueueEntries', N'U') IS NULL
            OR NOT EXISTS (SELECT 1 FROM dbo.QueueEntries q WHERE q.ShopId = s.Id)
          )
      AND (
            OBJECT_ID(N'dbo.Employee', N'U') IS NULL
            OR NOT EXISTS (SELECT 1 FROM dbo.Employee e WHERE e.ShopId = s.Id)
          );

    SET IDENTITY_INSERT dbo.Shops ON;
    INSERT INTO dbo.Shops (Id, Name, AdminId, OpeningTime, ClosingTime, QueueLimit, LunchBreak, QrCode)
    SELECT
        x.id,
        x.Name,
        x.AdminId,
        x.OpeningTime,
        x.ClosingTime,
        ISNULL(x.QueueLimit, 0),
        x.LunchBreak,
        x.QrCode
    FROM dbo.Shop x
    WHERE NOT EXISTS (SELECT 1 FROM dbo.Shops s WHERE s.Id = x.id);
    SET IDENTITY_INSERT dbo.Shops OFF;
END

CREATE OR ALTER PROCEDURE dbo.CheckToken
    @Token NVARCHAR(50)
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @AccountId BIGINT = 0;
    SELECT @AccountId = AccountId
    FROM dbo.Token
    WHERE Token = @Token AND ValidTo > GETUTCDATE();
    SELECT ISNULL(@AccountId, 0);
END
GO

CREATE OR ALTER PROCEDURE dbo.AddAccessToken
    @AccountId BIGINT,
    @Token NVARCHAR(50)
AS
BEGIN
    SET NOCOUNT ON;
    INSERT INTO dbo.Token (AccountId, Token, ValidTo)
    VALUES (@AccountId, @Token, DATEADD(HOUR, 12, GETUTCDATE()));
END
GO

CREATE OR ALTER PROCEDURE dbo.ExtendAccessToken
    @Token NVARCHAR(50)
AS
BEGIN
    SET NOCOUNT ON;
    UPDATE dbo.Token
    SET ValidTo = DATEADD(HOUR, 12, GETUTCDATE())
    WHERE Token = @Token;
END
GO

/* QADB.CreateCustomerJoinQueue(@Name 200, @Email 255, @PhoneNumber 30, @ShopId)
   Returns UserId, QueueEntryId, CreatedUser, AlreadyWaiting */
CREATE OR ALTER PROCEDURE dbo.CreateCustomerJoinQueue
    @Name NVARCHAR(200),
    @Email NVARCHAR(255),
    @PhoneNumber NVARCHAR(30),
    @ShopId INT
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    BEGIN TRY
        BEGIN TRANSACTION;

        DECLARE @UserId INT;
        DECLARE @AlreadyWaiting BIT = 0;
        DECLARE @CreatedUser BIT = 0;
        DECLARE @QueueEntryId INT;
        DECLARE @QueueLimit INT = 0;
        DECLARE @WaitingNow INT = 0;
        DECLARE @ShopExists BIT = 0;

        SELECT @QueueLimit = QueueLimit, @ShopExists = 1 FROM dbo.Shops WHERE Id = @ShopId;
        IF @ShopExists = 0
        BEGIN
            RAISERROR(N'Shop not found.', 16, 1);
        END

        SELECT @UserId = Id
        FROM dbo.Users
        WHERE Email = @Email AND Role = N'Customer' AND Deleted = 0;

        IF @UserId IS NULL
        BEGIN
            INSERT INTO dbo.Users (Name, Email, PhoneNumber, ShopId, Role, Created, Deleted)
            VALUES (@Name, @Email, @PhoneNumber, @ShopId, N'Customer', SYSUTCDATETIME(), 0);
            SET @UserId = CAST(SCOPE_IDENTITY() AS INT);
            SET @CreatedUser = 1;
        END
        ELSE
        BEGIN
            UPDATE dbo.Users
            SET Name = @Name,
                PhoneNumber = @PhoneNumber,
                ShopId = @ShopId
            WHERE Id = @UserId;
        END

        IF EXISTS (
            SELECT 1 FROM dbo.QueueEntries
            WHERE UserId = @UserId AND ShopId = @ShopId AND Status = 1
        )
        BEGIN
            SET @AlreadyWaiting = 1;
            SELECT TOP 1 @QueueEntryId = Id
            FROM dbo.QueueEntries
            WHERE UserId = @UserId AND ShopId = @ShopId AND Status = 1
            ORDER BY JoinedAt DESC;
        END
        ELSE
        BEGIN
            SELECT @WaitingNow = COUNT(*)
            FROM dbo.QueueEntries
            WHERE ShopId = @ShopId AND Status = 1;

            IF @QueueLimit > 0 AND @WaitingNow >= @QueueLimit
            BEGIN
                RAISERROR(N'This shop''s queue is full right now.', 16, 1);
            END

            INSERT INTO dbo.QueueEntries (UserId, ShopId, Status)
            VALUES (@UserId, @ShopId, 1);
            SET @QueueEntryId = CAST(SCOPE_IDENTITY() AS INT);
        END

        COMMIT TRANSACTION;

        SELECT
            @UserId AS UserId,
            @QueueEntryId AS QueueEntryId,
            @CreatedUser AS CreatedUser,
            @AlreadyWaiting AS AlreadyWaiting;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
        THROW;
    END CATCH
END
GO

CREATE OR ALTER FUNCTION dbo.QueuePosition(@QueueEntryId INT)
RETURNS INT
AS
BEGIN
    DECLARE @ShopId INT, @JoinedAt DATETIME2, @Status INT, @Pos INT;
    SELECT @ShopId = ShopId, @JoinedAt = JoinedAt, @Status = Status
    FROM dbo.QueueEntries WHERE Id = @QueueEntryId;
    IF @Status <> 1 RETURN NULL;
    SELECT @Pos = COUNT(*) + 1
    FROM dbo.QueueEntries
    WHERE ShopId = @ShopId AND Status = 1 AND JoinedAt < @JoinedAt;
    RETURN @Pos;
END
GO

/* QADB.QueueEntriesGet(@ShopId, @Status) — @Status 0 = waiting+serving */
CREATE OR ALTER PROCEDURE dbo.QueueEntriesGet
    @ShopId INT,
    @Status INT = 0
AS
BEGIN
    SET NOCOUNT ON;
    SELECT
        q.Id, q.UserId, q.ShopId, q.JoinedAt, q.ServedAt, q.Status,
        qs.Status AS StatusName,
        u.Name, u.Email, u.PhoneNumber,
        s.Name AS ShopName,
        CASE WHEN q.Status = 1 THEN dbo.QueuePosition(q.Id) ELSE NULL END AS Position
    FROM dbo.QueueEntries q
    INNER JOIN dbo.Users u ON u.Id = q.UserId
    INNER JOIN dbo.Shops s ON s.Id = q.ShopId
    INNER JOIN dbo.QueueStatus qs ON qs.Id = q.Status
    WHERE q.ShopId = @ShopId
      AND (
            (@Status > 0 AND q.Status = @Status)
            OR (@Status = 0 AND q.Status IN (1, 2))
          )
    ORDER BY CASE WHEN q.Status = 2 THEN 0 ELSE 1 END, q.JoinedAt;
END
GO

CREATE OR ALTER PROCEDURE dbo.QueueEntriesGetHistory
    @ShopId INT
AS
BEGIN
    SET NOCOUNT ON;
    SELECT
        q.Id, q.UserId, q.ShopId, q.JoinedAt, q.ServedAt, q.Status,
        qs.Status AS StatusName,
        u.Name, u.Email, u.PhoneNumber,
        s.Name AS ShopName,
        CAST(NULL AS INT) AS Position
    FROM dbo.QueueEntries q
    INNER JOIN dbo.Users u ON u.Id = q.UserId
    INNER JOIN dbo.Shops s ON s.Id = q.ShopId
    INNER JOIN dbo.QueueStatus qs ON qs.Id = q.Status
    WHERE q.ShopId = @ShopId AND q.Status IN (3, 4)
    ORDER BY q.JoinedAt DESC;
END
GO

CREATE OR ALTER PROCEDURE dbo.QueueEntriesUpdateStatus
    @Id INT,
    @Status INT
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @Current INT;
    SELECT @Current = Status FROM dbo.QueueEntries WHERE Id = @Id;
    IF @Current IS NULL
    BEGIN
        RAISERROR(N'Queue entry not found.', 16, 1);
        RETURN;
    END
    IF @Status = 2 AND @Current <> 1
    BEGIN
        RAISERROR(N'Only waiting customers can be taken to the chair.', 16, 1);
        RETURN;
    END
    IF @Status = 3 AND @Current NOT IN (1, 2)
    BEGIN
        RAISERROR(N'That visit is already closed.', 16, 1);
        RETURN;
    END

    UPDATE dbo.QueueEntries
    SET Status = @Status,
        ServedAt = CASE WHEN @Status = 3 THEN SYSUTCDATETIME() ELSE ServedAt END
    WHERE Id = @Id;

    EXEC dbo.QueueEntriesGetById @Id = @Id;
END
GO
